% Centred name, headline and contact lines; the photo, if any, centred above them.
\newcommand{\makecvtitle}{%
  \begingroup\centering
  \ifdefined\jaaphoto\jaaphoto\par\vspace{6pt}\fi
  {\fontsize{24pt}{28pt}\selectfont\jaafirst\ifdefempty{\jaalast}{}{\space\jaalast}\par}
  \ifdefined\jaaheadline\vspace{3pt}{\large\itshape\color{jaaaccent}\jaaheadline\par}\fi
  \vspace{6pt}
  {\small\jaacontact\par\ifdefined\jaaextra\jaaextra\par\fi}
  \ifdefined\jaaquote
    \vspace{8pt}\begin{minipage}{0.8\linewidth}\centering\itshape\jaaquote\end{minipage}\par
  \fi
  \endgroup}

% Text in the left margin column: dates of an entry, a language's name.
\newcommand{\jaamargin}[1]{%
  \noindent\llap{\parbox[t]{\jaadatecol}{\raggedright\small\leavevmode{\color{jaamuted}#1}}}}

% \cventry{dates}{title}{organization}{location}{}{description}: dates in the margin column,
% the title in bold, then organization and location in italics, then the description. The
% body is indented by the column's width, so long entries break across pages like any text.
% The organization comes as \textsc{...}, printed as written: small caps extract as capitals
% with stray spaces.
\newcommand{\cventry}[6]{%
  \par\needspace{3\baselineskip}\addvspace{6pt}%
  \begingroup\leftskip=\jaadatecol\renewcommand{\textsc}[1]{##1}
  \jaamargin{#1}{\bfseries #2}\par
  \ifblank{#3#4}{}{{\itshape #3\ifblank{#3}{}{\ifblank{#4}{}{, }}#4}\par}%
  \ifblank{#6}{}{#6\par}%
  \endgroup}
\newcommand{\cvitem}[2]{%
  \par\addvspace{2pt}\begingroup\leftskip=\jaadatecol\jaamargin{#1}#2\par\endgroup}
\newcommand{\cvlistitem}[1]{%
  \par\addvspace{1pt}\begingroup\leftskip=\dimexpr\jaadatecol+1em\relax
  \hangindent=0.8em\hangafter=1\makebox[0.8em][l]{\textbullet}#1\par\endgroup}
\newcommand{\cvlanguage}[3]{%
  \par\addvspace{2pt}\begingroup\leftskip=\jaadatecol
  \jaamargin{#1}#2\ifblank{#3}{}{ (#3)}\par\endgroup}
