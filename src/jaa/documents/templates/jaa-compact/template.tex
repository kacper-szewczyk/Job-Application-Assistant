% Name and headline on the left, contact details right-aligned beside them (and the photo, if
% any, at the right edge). The left block comes first in the PDF, so text extraction reads the
% name before the contact details.
\newcommand{\makecvtitle}{%
  \begin{minipage}[t]{0.55\linewidth}\raggedright
    {\fontsize{24pt}{27pt}\selectfont\bfseries
     \jaafirst\ifdefempty{\jaalast}{}{\space\jaalast}\par}
    \ifdefined\jaaheadline\vspace{3pt}{\large\color{jaaaccent}\jaaheadline\par}\fi
  \end{minipage}%
  \hfill
  \begin{minipage}[t]{\dimexpr0.45\linewidth\ifdefined\jaaphoto-72pt\fi\relax}
    \raggedleft\small\color{jaamuted}\jaacontact
  \end{minipage}%
  \ifdefined\jaaphoto\hspace{8pt}\raisebox{\dimexpr-\height+\ht\strutbox\relax}{\jaaphoto}\fi
  \par
  \ifdefined\jaaquote\vspace{6pt}{\small\jaaquote\par}\fi}

% \cventry{dates}{title}{organization}{location}{}{description}: one line with the title,
% organization and location, and the dates flush right (on the next line when they do not
% fit). The organization comes as \textsc{...}, printed as written: TeX Gyre has no real small
% caps, and faked ones extract as capitals with stray spaces.
\newcommand{\cventry}[6]{%
  \par\needspace{2\baselineskip}\addvspace{4pt}%
  \begingroup\renewcommand{\textsc}[1]{##1}%
  {\bfseries #2}%
  \ifblank{#3}{}{\ifblank{#2}{}{,\space}#3}%
  \ifblank{#4}{}{\space{\color{jaamuted}\textendash\space#4}}%
  \ifblank{#1}{}{\unskip\nobreak\hfil\penalty50\hskip1em\hbox{}\nobreak\hfil
    {\color{jaamuted}#1}}%
  {\rightskip=0pt\parfillskip=0pt\par}%
  \ifblank{#6}{}{{\small\color{jaamuted}#6}\par}%
  \endgroup}
