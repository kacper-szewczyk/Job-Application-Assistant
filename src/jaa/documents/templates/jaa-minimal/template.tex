% \cventry{dates}{title}{organization}{location}{}{description}: the title with the dates
% flush right (they move to the next line when both do not fit), then organization and
% location, then the description. needspace keeps an entry's first lines together. The
% organization comes as \textsc{...}: TeX Gyre Heros has no real small caps, and faked ones
% extract as capitals with stray spaces, so it is printed as written.
\newcommand{\cventry}[6]{%
  \par\begingroup\renewcommand{\textsc}[1]{##1}\needspace{3\baselineskip}\addvspace{5pt}%
  {\bfseries #2}%
  \ifblank{#1}{}{\unskip\nobreak\hfil\penalty50\hskip1em\hbox{}\nobreak\hfil
    {\small\color{jaamuted}#1}}%
  {\rightskip=0pt\parfillskip=0pt\par}%
  \ifblank{#3#4}{}{%
    {\color{jaamuted}#3\ifblank{#3}{}{\ifblank{#4}{}{, }}#4}\par}%
  \ifblank{#6}{}{{\small #6}\par}\endgroup}
\newcommand{\cvitem}[2]{%
  \par\addvspace{2pt}\ifblank{#1}{}{\textbf{#1}: }#2\par}
\newcommand{\cvlistitem}[1]{%
  \par\addvspace{1pt}\leftskip=1em\hangindent=0.9em\hangafter=1
  \makebox[0.9em][l]{\textbullet}#1\par\leftskip=0pt}
\newcommand{\cvlanguage}[3]{%
  \par\addvspace{2pt}\textbf{#1}\ifblank{#2}{}{ -- #2}\ifblank{#3}{}{ (#3)}\par}
